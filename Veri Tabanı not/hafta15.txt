procedure olusturma

create procedure hizala
as
begin
   SELECT   verilen_urun.musteri_kodu, sum(para.verilen_para), sum(verilen_urun.urun_tutar)
   FROM     para
   INNER JOIN verilen_urun
   ON para.musteri_kodu = verilen_urun.musteri_kodu
   group by verilen_urun.musteri_kodu
   order by sum (para.verilen_para) desc
end

create procedure hizala2
@musteri_kodu int
as
 begin 
SELECT   verilen_urun.musteri_kodu, sum(para.verilen_para), sum(verilen_urun.urun_tutar)
   FROM     para
   INNER JOIN verilen_urun
   ON para.musteri_kodu = verilen_urun.musteri_kodu
   where para.musteri_kodu=@musteri_kodu
   group by verilen_urun.musteri_kodu
   order by sum (para.verilen_para) desc
   end
