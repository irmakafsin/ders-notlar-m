import requests
from bs4 import BeautifulSoup
def get_url(url):
    r = requests.get(url)
    soup = BeautifulSoup(r.content,'html.parser')
    list2 = soup.find_all('div',{"class": "col-lg-6"})
    for i in list2:
        for link in i.find_all('a'):
            my_link = link.get('href')+"\n"
        yeni_link = '{}{}'.format("https:/www.milligazete.com.tr/arsiv" , my_link)
        print(yeni_link)
        with open ('irmak.txt'"a", encoding="utf-8") as file:
         file.write(yeni_link)
